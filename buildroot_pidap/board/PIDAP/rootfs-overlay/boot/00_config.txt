kernel=zImage

# Core hardware interfaces
dtparam=i2c_arm=on
dtparam=spi=on
dtparam=i2s=on

# Disable on-board audio; enable Hifiberry DAC
dtparam=audio=off
dtoverlay=hifiberry-dac

# Pirate Audio headphone amp enable
gpio=25=op,dh


# Automatically load overlays for detected DSI displays
display_auto_detect=1

# Automatically load initramfs files, if found
auto_initramfs=1

# Enable DRM VC4 V3D driver
dtoverlay=vc4-kms-v3d,noaudio
max_framebuffers=2

# Don't have the firmware create an initial video= setting in cmdline.txt.
# Use the kernel's default instead.
disable_fw_kms_setup=1

# Disable compensation for displays with overscan
disable_overscan=0


# Pirate Audio 240x240 ST7789 TFT on SPI0 (fbtft st7789v)
## dtoverlay=pirate-tft

dtoverlay=mipi-dbi-spi,spi0-1,speed=48000000,write-only,dc-gpio=9,reset-gpio=27,backlight-gpio=13,width=240,height=240
dtparam=width=240,height=240

hdmi_blanking=2
hdmi_force_hotplug=0
hdmi_ignore_hotplug=1


# USB host for the Zero
dtoverlay=dwc2,dr_mode=host
