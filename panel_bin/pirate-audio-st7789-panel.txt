# Pirate Audio 240x240 ST7789 init sequence for panel-mipi-dbi-spi
# Derived from the Pimoroni st7789 Python library.
# Use with: python3 make_panel_bin.py pirate-audio-st7789-panel.txt panel.bin

# Software reset
delay 10
command 0x01
delay 150

# MADCTL: orientation/mirroring. 0x70 is the only value that renders cleanly
# on this ST7789 with panel-mipi-dbi-spi. Screen rotation is done with fbcon.
command 0x36 0x70

# Frame rate control - idle mode
command 0xB2 0x0C 0x0C 0x00 0x33 0x33

# COLMOD: 16-bit/pixel (RGB565)
command 0x3A 0x05

# Gate control
command 0xB7 0x14

# VCOMS
command 0xBB 0x37

# LCM control
command 0xC0 0x2C

# VDV/VRHEN
command 0xC2 0x01

# VRHS
command 0xC3 0x12

# VDVS
command 0xC4 0x20

# Power control 1
command 0xD0 0xA4 0xA1

# Frame rate control - normal mode
command 0xC6 0x0F

# Positive gamma
command 0xE0 0xD0 0x04 0x0D 0x11 0x13 0x2B 0x3F 0x54 0x4C 0x18 0x0D 0x0B 0x1F 0x23

# Negative gamma
command 0xE1 0xD0 0x04 0x0C 0x11 0x13 0x2C 0x3F 0x44 0x51 0x2F 0x1F 0x1F 0x20 0x23

# Inversion on (set to 0x20 if colours look inverted)
command 0x21

# Sleep out
command 0x11

# Display on
command 0x29
delay 100
